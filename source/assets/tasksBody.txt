  ensure "is debian" {                # run a shell command and check it

      output = "debian"               # string | { contains = "deb" } | { matches = "^debian.*$" }

  http http://localhost:8080/health { # submit a query and test the answer (a check, like ensure, runs even in check mode, never changes)
      type = "GET"                    # type is any HTTP method (default GET)
      headers = ["X-Api-Key=k"]       # headers are "Name=Value" strings
      data = "{\"x\": 1}"             # data is the body sent verbatim
      code = 200                      # code (default 200) tests the status
      output = { contains = "ok" }    # output the body (ensure's shapes)
      timeout = 10                    # timeout bounds the whole query (default 10s).
                                      # Plain http only, queried by tachy itself (no curl needed)

  repo /srv/app {                     # ensure a git repository exists and is
